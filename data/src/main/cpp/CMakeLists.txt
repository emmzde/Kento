cmake_minimum_required(VERSION 3.22)
project(kento_speech LANGUAGES C CXX)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
# x86_64 runs on emulators and Intel Chromebooks; without vector instructions speech is several times slower.
# OfflineSpeech checks /proc/cpuinfo before loading, so a CPU without AVX2 gets an error instead of SIGILL.
if(ANDROID_ABI STREQUAL "x86_64")
    set(KENTO_VECTOR ON)
else()
    set(KENTO_VECTOR OFF)
endif()
set(GGML_AVX ${KENTO_VECTOR} CACHE BOOL "" FORCE)
set(GGML_AVX2 ${KENTO_VECTOR} CACHE BOOL "" FORCE)
set(GGML_FMA ${KENTO_VECTOR} CACHE BOOL "" FORCE)
set(GGML_F16C ${KENTO_VECTOR} CACHE BOOL "" FORCE)
# Run inference at normal speed in QA and debug builds as well.
add_compile_options(-O2)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../build/whisper-source/whisper.cpp-1.7.6" whisper EXCLUDE_FROM_ALL)
add_library(kento_speech SHARED speech.cpp)
target_compile_features(kento_speech PRIVATE cxx_std_17)
target_link_libraries(kento_speech PRIVATE whisper android log)
target_link_options(kento_speech PRIVATE "-Wl,-z,max-page-size=16384")
